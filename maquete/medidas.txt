# MEDIDAS DA MAQUETE (mesas, cadeiras e balcão para imprimir em 3D). mm.
#
#   1. Mude o número depois do "=". Pode usar vírgula ou ponto.
#   2. Salve o arquivo (Ctrl+S).
#   3. Dê dois cliques em GERAR.bat.
#   4. Os arquivos novos aparecem em stl\ (impressora) e step\ (Fusion 360).
#      step\maquete_completa.step = o salão inteiro montado, com o robô.
#
# Tudo depois de "#" é comentário e não conta.
#
# Proporção (escala ~1 : 4,6 do Bandejito, que tem ~27 cm):
#   mesa real 75 cm de altura e tampo de 80 cm -> 15 cm e 18 cm
#   cadeira real: assento a 45 cm, encosto até 90 cm -> 10 cm e ~19 cm
#   balcão real: ~80 cm -> ~17 cm (um pouco acima da bandeja do robô)
# =====================================================================

# --- Mesa ---------------------------------------------------------------

mesa_altura        = 150     # do chão até o topo do tampo
tampo_esp          = 4       # espessura do tampo
tampo_raio_canto   = 12      # só para a quadrada: arredondado dos cantos
numero_tamanho     = 60      # altura do número gravado no tampo (0 = sem número)
numero_prof        = 0.8     # profundidade da gravação
perna_d            = 24      # diâmetro da perna da mesa
pe_d               = 110     # diâmetro do pé (base redonda no chão)

encaixe_prof       = 10      # quanto a perna entra no tampo e no pé

# --- Balcão (oco e ripado, em módulos iguais) -----------------------------
balcao_modulos     = 2       # quantos módulos lado a lado (imprima esse tanto)

ripa_largura       = 8       # ripas da frente e dos lados
ripa_vao           = 5       # fresta entre as ripas (0 = parede lisa; maior = gasta menos)

# --- Impressão ------------------------------------------------------------
